# DEPENDÊNCIAS DO PROJETO HÓRUS AI
#
# Política de versão: piso >= na versão que corrige a última CVE conhecida,
# teto < no próximo major. Assim o projeto continua recebendo correções de
# segurança (patches e minors) sem receber uma major nova que quebre a API
# sem aviso. Instale/atualize com:
#     pip install -r requirements.txt

# --- Framework Web e Segurança ---
flask>=3.1.3,<4          # >=3.1.3 corrige CVE-2026-27205 (cache poisoning via sessao)
flask-cors>=6.0.5,<7     # >=5.0.1 corrige CVE-2024-6221/1681/6839/6844/6866
flask-limiter>=4.1.1,<5  # 4.x e a linha suportada (3.8.0 estava 2 majors atras)
werkzeug>=3.0.6,<4       # >=3.0.6 corrige CVE-2024-49766/49767

# --- Inteligência Artificial ---
groq>=1.6.0,<2           # API usada aqui (Groq(api_key=...) + chat.completions.create)

# --- Configurações e Ambiente ---
python-dotenv>=1.2.2,<2  # >=1.2.2 corrige CVE-2026-28684 (set_key/unset_key seguem symlink)

# --- Servidor de Produção ---
# O gunicorn depende de fcntl, que so existe em Unix: ele instala no Windows
# mas nao roda. O marcador evita instalar um pacote inutil na maquina de
# desenvolvimento Windows, sem deixar de instala-lo no servidor Linux.
gunicorn>=26.0.0,<27; sys_platform != "win32"   # >=23.0.0 corrige CVE-2024-6827 (request smuggling)
